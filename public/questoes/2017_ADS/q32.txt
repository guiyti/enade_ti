Iom base nas informações e na situação apresentada,  qual o comando ^Y> que seleciona corretamente 
os nomes dos candidatos,  seus partidos e o total de votos de cada partido nessa eleição͍
A SELECT c.nome, p.nome, SUM(v.votos) FROM Partido p, Candidato c, 
Votacao v WHERE c.numero = p.numero and v.partido = c.numero;
B SELECT c.nome, p.nome, COUNT(v.votos) FROM Partido p, Candidato c, 
Votacao v WHERE c.numero = p.numero and v.partido = c.numero GROUP BY 
c.nome, p.nome;
C SELECT c.nome, p.nome, SUM(v.votos) FROM Partido p, Candidato c, 
Votacao v WHERE c.numero = p.numero and v.partido = c.numero GROUP BY 
c.nome, p.nome;
D SELECT c.nome, p.nome, v.votos FROM Partido p, Candidato c, Votacao v 
WHERE c.numero = p.numero and v.partido = c.numero GROUP BY c.nome, 
p.nome, SUM(v.votos);
